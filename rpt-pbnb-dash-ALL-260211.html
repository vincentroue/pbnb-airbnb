corr = {
  const x = filtered.map(d => d[xMetric.value]);
  const y = filtered.map(d => d[yMetric.value]);
  const n = x.length;
  const mx = d3.mean(x), my = d3.mean(y);
  const num = d3.sum(x.map((xi, i) => (xi - mx) * (y[i] - my)));
  const den = Math.sqrt(d3.sum(x.map(xi => (xi - mx)**2)) * d3.sum(y.map(yi => (yi - my)**2)));
  return den === 0 ? 0 : num / den;
}
